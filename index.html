// KSHATRIYA SEOS // CORE V3.0 - FRONTEND INTERFACE const WORKER_URL = "https://kshatriya-ai-bridge.solankipriyam71.workers.dev"; async function sendMessage() { const inputField = document.getElementById("user-input"); const chatContainer = document.getElementById("chat-container"); if (!inputField || !inputField.value.trim()) return; const userPrompt = inputField.value.trim(); // Append User Message to UI appendMessage("User", userPrompt); inputField.value = ""; // Clear input try { // Core Network Execution const response = await fetch(WORKER_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ prompt: userPrompt }) }); if (!response.ok) { throw new Error(`Network alert: ${response.status}`); } const data = await response.json(); // Output Parsing & Verification if (data && data.reply) { appendMessage("Core", data.reply); } else { appendMessage("Core", "Error: Frame empty or invalid response format."); } } catch (error) { console.error("Pipeline Breakdown:", error); appendMessage("Core", `System Error: ${error.message}`); } } // UI Rendering Engine function appendMessage(sender, text) { const chatContainer = document.getElementById("chat-container"); if (!chatContainer) return; const messageDiv = document.createElement("div"); messageDiv.className = sender === "User" ? "user-msg-block" : "core-msg-block"; // Handles direct HTML injection for error tracking/formatting messageDiv.innerHTML = `> ${sender}:
${text}`; chatContainer.appendChild(messageDiv); chatContainer.scrollTop = chatContainer.scrollHeight; // Auto-scroll } // Event Binding for Execution document.getElementById("exec-btn")?.addEventListener("click", sendMessage); document.getElementById("user-input")?.addEventListener("keypress", (e) => { if (e.key === "Enter") sendMessage(); });
